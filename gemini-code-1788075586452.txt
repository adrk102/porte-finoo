        function handleGalleryFileChange(e, prodId, imgIdx) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(evt) {
                const base64 = evt.target.result;
                const p = [...cadenas, ...pulseras].find(item => item.id === prodId);
                if (p) {
                    p.images[imgIdx] = base64;
                    saveCatalogs(false);
                    openGalleryManager(prodId);
                    showToast(`Foto ${imgIdx + 1} actualizada con éxito`);
                }
            };
            reader.readAsDataURL(file);
        }

        function setActiveModalImage(id, idx) {
            const p = [...cadenas, ...pulseras].find(item => item.id === id);
            if (!p || !p.images[idx]) return;
            currentModalImageIndex = idx;
            document.getElementById('modal-main-img').src = p.images[idx];
            document.getElementById('modal-idx-txt').innerText = idx + 1;
        }

        function nextModalImg(id) {
            let next = currentModalImageIndex + 1;
            if (next >= 5) next = 0;
            setActiveModalImage(id, next);
        }

        function prevModalImg(id) {
            let prev = currentModalImageIndex - 1;
            if (prev < 0) prev = 4;
            setActiveModalImage(id, prev);
        }

            modalQuantity += delta;
            if (modalQuantity < 1) modalQuantity = 1;
            document.getElementById('modal-qty-val').innerText = modalQuantity;

        function quickAddToCart(id) {
            addToCart(id, 1);

            addToCart(id, modalQuantity);
            closeProductModal();

        function addToCart(id, qty) {
            const p = [...cadenas, ...pulseras].find(item => item.id === id);
            if (!p) return;

            const existing = cart.find(i => i.id === id);
            if (existing) existing.quantity += qty;
            else cart.push({ id: p.id, name: p.name, price: p.price, image: p.images[0], quantity: qty });

            updateCartUI();
            showToast(`"${p.name}" agregado al carrito`);
        }

        function updateCartUI() {
            const counter = document.getElementById('cart-counter');
            const container = document.getElementById('cart-items-container');
            const totalEl = document.getElementById('cart-total');

            const totalItems = cart.reduce((sum, i) => sum + i.quantity, 0);
            if (totalItems > 0) {
                counter.innerText = totalItems;
                counter.classList.remove('scale-0');
            } else {
                counter.classList.add('scale-0');
            }

            if (cart.length === 0) {
                container.innerHTML = '<p class="text-center text-brand-silver py-6 text-[10px]">Tu carrito está vacío</p>';
                totalEl.innerText = '$0 MXN';
                return;
            }

            let total = 0;
            container.innerHTML = cart.map(i => {
                total += i.price * i.quantity;
                return `
                    <div class="flex items-center gap-2 bg-brand-dark p-2 rounded border border-brand-grayMid text-[10px]">
                        <img src="${i.image}" class="w-9 h-9 object-cover rounded">
                        <div class="flex-1">
                            <h4 class="font-bold text-white">${i.name}</h4>
                            <span>${i.quantity}x - $${i.price * i.quantity} MXN</span>
                        </div>
                        <button onclick="removeFromCart(${i.id})" class="text-brand-silver hover:text-red-400 p-1"><i class="fa-solid fa-trash-can"></i></button>
                    </div>
                `;
            }).join('');

            totalEl.innerText = `$${total} MXN`;
        }

        function removeFromCart(id) {
            cart = cart.filter(i => i.id !== id);
            updateCartUI();
        }
